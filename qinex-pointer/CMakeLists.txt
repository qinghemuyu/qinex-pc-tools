cmake_minimum_required(VERSION 3.21)
project(QInEXPointer LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "QInEX Pointer 需要 Windows x64 和 Visual Studio C++。")
endif()
if(MSVC)
    add_compile_options(/utf-8 /W4 /permissive-)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
option(QINEX_BUILD_POINTER "构建独立 Pointer 程序" ON)
option(QINEX_WITH_ANDROID_TOOLS "包含 Android 手机工具及内嵌运行依赖" OFF)
add_compile_definitions(QINEX_WITH_ANDROID_TOOLS=$<BOOL:${QINEX_WITH_ANDROID_TOOLS}>)
add_library(pointer_core src/pointer_core.cpp)
target_include_directories(pointer_core PUBLIC src)
add_library(pointer_network src/udp_receiver.cpp)
target_include_directories(pointer_network PUBLIC src)
target_link_libraries(pointer_network PUBLIC ws2_32)
add_library(pointer_capture src/capture_output.cpp src/directshow_capture.cpp src/capture_diagnostics.cpp)
target_include_directories(pointer_capture PUBLIC src)
target_link_libraries(pointer_capture PUBLIC mfplat mfreadwrite mfuuid mf strmiids ole32 oleaut32 user32 gdi32)
add_library(pointer_presenter src/capture_presenter.cpp src/custom_cursor.cpp)
target_include_directories(pointer_presenter PUBLIC src)
target_link_libraries(pointer_presenter PUBLIC pointer_core pointer_capture d3d11 dxgi d2d1 dwrite gdiplus)
if(QINEX_WITH_ANDROID_TOOLS)
find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
set(QINEX_ADB_DIR "$ENV{ANDROID_HOME}/platform-tools" CACHE PATH "ADB 35.0.2 目录")
set(QINEX_ANDROID_BUILD_TOOLS "$ENV{ANDROID_HOME}/build-tools/34.0.0" CACHE PATH "内嵌 Android Build Tools 34.0.0")
set(QINEX_ANDROID_JDK "${CMAKE_CURRENT_SOURCE_DIR}/build/android-dependencies/jdk-17.0.20.1+1" CACHE PATH "用于 jlink 的 Temurin OpenJDK 17")
if(NOT EXISTS "${QINEX_ANDROID_JDK}/bin/jlink.exe")
    message(FATAL_ERROR "请运行 python tools/fetch_temurin.py 准备内嵌 OpenJDK 17，或设置 QINEX_ANDROID_JDK。")
endif()
set(QINEX_ANDROID_RUNTIME "${CMAKE_CURRENT_BINARY_DIR}/android-runtime")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_android_runtime.py"
    --adb "${QINEX_ADB_DIR}" --sdk "${QINEX_ANDROID_BUILD_TOOLS}" --jdk "${QINEX_ANDROID_JDK}" --output "${QINEX_ANDROID_RUNTIME}"
    RESULT_VARIABLE QINEX_RUNTIME_RESULT)
if(NOT QINEX_RUNTIME_RESULT EQUAL 0)
    message(FATAL_ERROR "Android 内嵌依赖准备失败，详见上方错误。")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS tools/prepare_android_runtime.py tools/android_runtime_versions.json)
add_library(pointer_android src/tool_process.cpp src/embedded_runtime.cpp src/adb_client.cpp
    src/android_tools_service.cpp src/android_cursor_package.cpp)
target_include_directories(pointer_android PUBLIC src PRIVATE "${QINEX_ANDROID_RUNTIME}")
target_link_libraries(pointer_android PUBLIC pointer_presenter bcrypt shell32 ole32 ws2_32 iphlpapi)
endif()
add_library(pointer_desktop STATIC src/main.cpp src/app_ui.cpp src/ui_theme.cpp src/ui_motion.cpp src/calibration.cpp src/desktop_helpers.cpp src/customization_dialogs.cpp src/demo_sender.cpp)
target_include_directories(pointer_desktop PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(QINEX_WITH_ANDROID_TOOLS)
    target_sources(pointer_desktop PRIVATE src/android_tools_page.cpp)
    # 资源直接进入最终 EXE，避免静态库中的资源被链接器丢弃。
    target_sources(pointer_desktop INTERFACE "${QINEX_ANDROID_RUNTIME}/runtime.rc")
    target_link_libraries(pointer_desktop PUBLIC pointer_android)
endif()
target_link_libraries(pointer_desktop PUBLIC pointer_core pointer_network pointer_capture pointer_presenter user32 gdi32 gdiplus comctl32 comdlg32 shell32 dwmapi iphlpapi uxtheme)
target_include_directories(pointer_desktop INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(QINEX_BUILD_POINTER)
add_executable(QInEX-Pointer WIN32 src/standalone.cpp src/app.rc)
target_link_libraries(QInEX-Pointer PRIVATE pointer_desktop)
if(MSVC)
    target_link_options(QInEX-Pointer PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
endif()
add_executable(QInEX-Pointer-Simulator tools/simulator.cpp)
target_link_libraries(QInEX-Pointer-Simulator PRIVATE ws2_32)
# 发布包只包含用户运行所需文件；模拟器和开发文档保留在源码与构建目录。
install(TARGETS QInEX-Pointer RUNTIME DESTINATION .)
install(FILES 使用说明.txt DESTINATION .)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE.txt" "${CMAKE_CURRENT_SOURCE_DIR}/../NOTICE.txt"
    "${CMAKE_CURRENT_SOURCE_DIR}/../THIRD_PARTY_NOTICES.md" DESTINATION .)
endif() # QINEX_BUILD_POINTER
include(CTest)
if(BUILD_TESTING)
    add_executable(pointer_runtime_tests tests/runtime_tests.cpp src/app.rc)
    target_link_libraries(pointer_runtime_tests PRIVATE pointer_desktop)
    # 配置按 EXE 目录读取，测试与用户运行目录隔离。
    set_target_properties(pointer_runtime_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime-test")
    target_link_options(pointer_runtime_tests PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
    add_test(NAME pointer_embedded_runtime COMMAND pointer_runtime_tests)
    set_tests_properties(pointer_embedded_runtime PROPERTIES TIMEOUT 30)
    add_executable(pointer_core_tests tests/pointer_core_tests.cpp)
    target_link_libraries(pointer_core_tests PRIVATE pointer_core)
    add_test(NAME pointer_core_behavior COMMAND pointer_core_tests)
    add_executable(pointer_capture_diagnostics_tests tests/capture_diagnostics_tests.cpp)
    target_link_libraries(pointer_capture_diagnostics_tests PRIVATE pointer_capture)
    add_test(NAME pointer_capture_diagnostics COMMAND pointer_capture_diagnostics_tests)
    if(QINEX_WITH_ANDROID_TOOLS)
    add_executable(pointer_android_tests tests/android_tools_tests.cpp "${QINEX_ANDROID_RUNTIME}/runtime.rc")
    target_link_libraries(pointer_android_tests PRIVATE pointer_android)
    add_test(NAME pointer_android_behavior COMMAND pointer_android_tests)
    endif()
endif()
