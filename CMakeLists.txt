cmake_minimum_required(VERSION 3.21)
project(QInEXPCTools LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "请使用 Windows x64 和 Visual Studio C++ 工具链。")
endif()
include(CTest)
option(QINEX_BUILD_LINK "构建 QInEX-Link" ON)
option(QINEX_BUILD_POINTER "构建 QInEX Pointer" ON)
option(QINEX_WITH_ANDROID_TOOLS "在 Pointer 中包含 Android 手机工具" OFF)
if(QINEX_BUILD_LINK)
    add_subdirectory(qinex-link)
endif()
if(QINEX_BUILD_POINTER)
    add_subdirectory(qinex-pointer)
endif()
